01 - Forecasting live births

This script aims to describe the process of applying the method for forecasting the number of live births (LB) in each health macro-region in Brazil. To do this, we will use data from the Live Births System (SINASC, Sistema de Nascidos Vivos) provided by the SUS Information Technology Department (DATASUS) for each health macro-region.

The SQL code used to create the query 'SELECT * FROM "Analytics Layer"."Epidemiológico".Nascimentos."Nascidos vivos por macrorregião de saúde"' is presented below.


WITH nascimentos_final 
AS (
    SELECT 
        SUBSTR(CODMUNRES, 1, 6) AS CODUFMUN,
        SUBSTR(DTNASC, 5, 4) AS ANO,
        SUBSTR(DTNASC, 3, 2) AS MES,
        COUNT(*) AS QUANTIDADE

    FROM Dados.sinasc.DN
    GROUP BY
        SUBSTR(CODMUNRES, 1, 6),
        SUBSTR(DTNASC, 5, 4),
        SUBSTR(DTNASC, 3, 2)
    ORDER BY
        ANO, MES),

nascimentos_preliminar 
AS (
    SELECT 
        SUBSTR(CODMUNRES, 1, 6) AS CODUFMUN,
        SUBSTR(DTNASC, 5, 4) AS ANO,
        SUBSTR(DTNASC, 3, 2) AS MES,
        COUNT(*) AS QUANTIDADE
    FROM 
        Dados.sinasc.preliminar
    WHERE 
      SUBSTR(DTNASC, 5, 4) > (SELECT MAX(SUBSTR(DTNASC, 5, 4)) 
      FROM Dados.sinasc.DN)
    GROUP BY
        SUBSTR(CODMUNRES, 1, 6),
        SUBSTR(DTNASC, 5, 4),
        SUBSTR(DTNASC, 3, 2))

SELECT
    m.regiao,
    m.uf_sigla,
    m.cod_macrorregiao,
    m.macrorregiao,
    dn.ano,
    dn.mes,
    sum(quantidade) as quantidade
FROM
  (SELECT * FROM nascimentos_final
   UNION ALL
   SELECT * FROM nascimentos_preliminar) dn
LEFT JOIN "Analytics Layer".Territorial."Municípios - Hierarquia Completa" m 
  ON CAST(dn.CODUFMUN AS INT) = m.cod_municipio
WHERE uf_sigla IS NOT NULL
GROUP BY 
    m.regiao,
    m.uf_sigla,
    m.cod_macrorregiao,
    m.macrorregiao,
    dn.ano,
    dn.mes
ORDER BY
    ANO, MES

1) Time split

Let’s do a time split to divide the data set into two subsets: training and test.

  • Training is a subset that supplies the algorithms;

  • Test is a subset used to assess how well the algorithm’s predictive output adheres to real values.

80% of the data set will be used for training and 20% for testing.

lb5210 <- lb |> filter(mc_code == "5210")

2) Fit models

We built an object containing various forecasting models, including Autoregressive Integrated Moving Average (ARIMA), ARIMA with XGBoost, Prophet, Exponential Smoothing (ETS) and Multivariate Adaptive Regression Splines (MARS).

# Model 1: auto_arima ----
model_fit_arima_no_boost <-
    arima_reg() |>
    set_engine(engine = "auto_arima") |>
    fit(qtt ~ date, data = training(splits))


# Model 2: arima_boost ----

model_fit_arima_boosted <- 
  arima_boost(
    min_n = 2,
    learn_rate = 0.015)  |> 
    set_engine(engine = "auto_arima_xgboost") |> 
    fit(qtt ~ date + 
              as.numeric(date) + 
              factor(month(date, 
                           label = TRUE),
                     ordered = F),
        data = training(splits))

# Model 3: ets ----
model_fit_ets <- 
  exp_smoothing() |>
    set_engine(engine = "ets") |>
    fit(qtt ~ date, data = training(splits))

# Model 4: prophet ----
model_fit_prophet <- 
  prophet_reg() |>
    set_engine(engine = "prophet") |>
    fit(qtt ~ date, data = training(splits))


# Model 5: lm ----
model_fit_lm <- 
  linear_reg() |>
    set_engine("lm") |>
    fit(qtt ~ as.numeric(date) + 
                factor(month(date, 
                             label = TRUE), 
                       ordered = FALSE),
        data = training(splits))

# Model 6: earth ----

model_spec_mars <- 
  mars(mode = "regression") |>
    set_engine("earth") 

recipe_spec <- 
  recipe(qtt ~ date, data = training(splits)) |>
    step_date(date, features = "month", ordinal = FALSE) |>
    step_mutate(date_num = as.numeric(date)) |>
    step_normalize(date_num) |>
    step_rm(date)
  
wflw_fit_mars <- 
  workflow() |>
    add_recipe(recipe_spec) |>
    add_model(model_spec_mars) |>
    fit(training(splits))


models_tbl <- 
  modeltime_table(
    model_fit_arima_no_boost,
    model_fit_arima_boosted,
    model_fit_ets,
    model_fit_prophet,
    model_fit_lm,
    wflw_fit_mars)

B) Function to automate process

We created a function to apply and maintain the best trained model based on the experience expressed in section A.

best_model <- 
  
  function(mc_region){
    
    # split
    lb_region <- lb |> 
      filter(mc_code == mc_region)
    
    splits <- initial_time_split(lb_region, 
                                 prop = 0.80)
    
    # models 
    
      # Model 1: auto_arima ----
      model_fit_arima_no_boost <-
          arima_reg() |>
          set_engine(engine = "auto_arima") |>
          fit(qtt ~ date, data = training(splits))
      
  
      # Model 2: arima_boost ----
      model_fit_arima_boosted <- 
        arima_boost(
          min_n = 2,
          learn_rate = 0.015)  |> 
          set_engine(engine = "auto_arima_xgboost") |> 
          fit(qtt ~ date + 
                    as.numeric(date) + 
                    factor(month(date, 
                                 label = TRUE),
                           ordered = F),
              data = training(splits))
      
      # Model 3: ets ----
      model_fit_ets <- exp_smoothing() |>
          set_engine(engine = "ets") |>
          fit(qtt ~ date, data = training(splits))
      
      # Model 4: prophet ----
      model_fit_prophet <- prophet_reg() |>
          set_engine(engine = "prophet") |>
          fit(qtt ~ date, data = training(splits))
      
    
    # table with models
      
    models_tbl <- 
      modeltime_table(
        model_fit_arima_no_boost,
        model_fit_arima_boosted,
        model_fit_ets,
        model_fit_prophet)
    
    calibration_tbl <- 
      models_tbl |>
      modeltime_calibrate(new_data = testing(splits))
    
    # evaluating the best model 
    
    min_mape <- calibration_tbl |> 
                modeltime_accuracy() |> 
                slice_min(mape, n = 1)
    
    refit_tbl <- calibration_tbl |>
                    modeltime_refit(data = lb_region) |>
                    filter(.model_id == min_mape$.model_id)

C) Forecasting all regions

Using a looping structure, we automate the process using both functions from section B.

region_vector <- lb |> 
                  distinct(mc_code)  
    

region_vector <- region_vector$mc_code

df_total <- data.frame()

for(i in region_vector){
  
  df <- best_model(i)
  
  df_total <- rbind(df_total, df)
  
  grafico(df)
  
}
